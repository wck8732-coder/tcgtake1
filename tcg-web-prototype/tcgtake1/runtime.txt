OUT_DIR = "runtime_package"

os.makedirs(OUT_DIR, exist_ok=True)

master = pd.read_csv(MASTER_CSV)

records = []
for _, row in master.iterrows():
    rec = {}
    for col, val in row.items():
        if pd.isna(val):
            rec[col] = ""
        elif col == "version":
            rec[col] = str(val)
        elif col in ["card_uid", "cost", "set_number", "max_copies_per_deck"]:
            rec[col] = int(val)
        elif col in ["is_legendary", "is_token"]:
            rec[col] = bool(val)
        elif col in ["attack", "life"]:
            if val == "":
                rec[col] = ""
            elif isinstance(val, float) and float(val).is_integer():
                rec[col] = int(val)
            else:
                rec[col] = val
        else:
            rec[col] = val
    records.append(rec)

with open(os.path.join(OUT_DIR, "tcgtake1_master_card_database.master.json"), "w", encoding="utf-8") as f:
    json.dump(records, f, indent=2)

effect_registry_seed = []
for rec in records:
    txt = str(rec.get("rules_text", "")).strip()
    effects = []
    lower = txt.lower()
    if "dies" in lower and "each player loses 1 life" in lower:
        effects = ["EFF_BOTH_PLAYERS_LOSE_1_ON_DEATH"]
    elif "draw" in lower:
        effects = ["EFF_DRAW_GENERIC"]
    elif ("heal" in lower) or ("gain" in lower and "life" in lower):
        effects = ["EFF_HEAL_GENERIC"]
    elif "destroy" in lower:
        effects = ["EFF_DESTROY_GENERIC"]
    elif "sacrifice" in lower:
        effects = ["EFF_SACRIFICE_GENERIC"]
    elif "purge" in lower:
        effects = ["EFF_PURGE_GENERIC"]
    elif "create" in lower and "token" in lower:
        effects = ["EFF_CREATE_TOKEN_GENERIC"]

    if rec.get("keywords", ""):
        for k in str(rec["keywords"]).split(","):
            k = k.strip()
            if k:
                effects.append(f"KW_{k.upper().replace(' ', '_')}")

    effect_registry_seed.append({
        "card_id": rec["id"],
        "rules_text": txt,
        "effect_ids": effects
    })
